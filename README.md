<script>
const canvas = document.getElementById("game");
const ctx = canvas.getContext("2d");

const keys = {};

document.addEventListener("keydown", e => {
    keys[e.key.toLowerCase()] = true;

    if (
        ["arrowup", "arrowdown", "arrowleft", "arrowright", " "]
        .includes(e.key.toLowerCase())
    ) {
        e.preventDefault();
    }

    if (e.key.toLowerCase() === "r") {
        resetGame();
    }
});

document.addEventListener("keyup", e => {
    keys[e.key.toLowerCase()] = false;
});

let player;
let platforms;
let coins;
let enemies;
let score;
let gameOver;
let win;

function resetGame() {

    player = {
        x: 80,
        y: 400,
        width: 34,
        height: 44,
        vx: 0,
        vy: 0,
        speed: 5,
        jump: 12,
        grounded: false
    };

    platforms = [
        {x: 0,   y: 510, w: 1000, h: 50},
        {x: 150, y: 420, w: 140,  h: 20},
        {x: 360, y: 350, w: 150,  h: 20},
        {x: 590, y: 420, w: 150,  h: 20},
        {x: 790, y: 320, w: 130,  h: 20}
    ];

    coins = [
        {x: 210, y: 375, collected:false},
        {x: 420, y: 305, collected:false},
        {x: 650, y: 375, collected:false},
        {x: 845, y: 275, collected:false}
    ];

    enemies = [
        {x: 300, y: 470, w: 35, h: 35, vx: 2},
        {x: 550, y: 470, w: 35, h: 35, vx: -2},
        {x: 750, y: 470, w: 35, h: 35, vx: 2}
    ];

    score = 0;
    gameOver = false;
    win = false;
}

resetGame();

function collision(a, b) {

    return (
        a.x < b.x + b.w &&
        a.x + a.width > b.x &&
        a.y < b.y + b.h &&
        a.y + a.height > b.y
    );
}

function update() {

    if (gameOver || win) return;

    player.vx = 0;

    if (keys["arrowleft"] || keys["a"])
        player.vx = -player.speed;

    if (keys["arrowright"] || keys["d"])
        player.vx = player.speed;

    if (
        (keys["arrowup"] || keys["w"] || keys[" "]) &&
        player.grounded
    ) {
        player.vy = -player.jump;
        player.grounded = false;
    }

    player.vy += 0.55;

    player.x += player.vx;
    player.y += player.vy;

    if (player.x < 0)
        player.x = 0;

    if (player.x + player.width > canvas.width)
        player.x = canvas.width - player.width;

    player.grounded = false;

    for (const p of platforms) {

        if (
            player.x + player.width > p.x &&
            player.x < p.x + p.w &&
            player.y + player.height <= p.y + 15 &&
            player.y + player.height + player.vy >= p.y
        ) {

            player.y = p.y - player.height;
            player.vy = 0;
            player.grounded = true;
        }
    }

    for (const c of coins) {

        if (!c.collected) {

            const coinBox = {
                x: c.x,
                y: c.y,
                width: 22,
                height: 22
            };

            if (collision(player, coinBox)) {
                c.collected = true;
                score += 100;
            }
        }
    }

    for (const e of enemies) {

        e.x += e.vx;

        if (e.x < 0 || e.x + e.w > canvas.width)
            e.vx *= -1;

        if (collision(player, e)) {

            if (player.vy > 0 &&
                player.y + player.height - 8 < e.y) {

                e.x = -100;
                score += 200;
                player.vy = -8;

            } else {

                gameOver = true;
            }
        }
    }

    if (coins.every(c => c.collected)) {
        win = true;
    }

    if (player.y > canvas.height) {
        gameOver = true;
    }
}

function drawPlayer() {

    const x = Math.floor(player.x);
    const y = Math.floor(player.y);

    // hat
    ctx.fillStyle = "#0f172a";
    ctx.fillRect(x + 3, y, 28, 8);

    // head
    ctx.fillStyle = "#f5b183";
    ctx.fillRect(x + 8, y + 7, 22, 17);

    // shirt
    ctx.fillStyle = "#ef4444";
    ctx.fillRect(x + 5, y + 23, 26, 12);

    // overalls
    ctx.fillStyle = "#2563eb";
    ctx.fillRect(x + 9, y + 32, 17, 12);

    // shoes
    ctx.fillStyle = "#020617";
    ctx.fillRect(x + 2, y + 40, 13, 5);
    ctx.fillRect(x + 22, y + 40, 13, 5);

    // eyes
    ctx.fillStyle = "#020617";
    ctx.fillRect(x + 20, y + 12, 3, 3);
}

function drawEnemy(e) {

    ctx.fillStyle = "#7c3aed";

    ctx.fillRect(e.x, e.y + 8, e.w, e.h - 8);

    ctx.fillStyle = "#020617";

    ctx.fillRect(e.x + 7, e.y + 15, 5, 5);
    ctx.fillRect(e.x + 23, e.y + 15, 5, 5);

    ctx.fillRect(e.x + 5, e.y + e.h - 4, 10, 5);
    ctx.fillRect(e.x + 20, e.y + e.h - 4, 10, 5);
}

function drawCoin(c) {

    if (c.collected) return;

    ctx.fillStyle = "#facc15";

    ctx.beginPath();
    ctx.arc(c.x + 11, c.y + 11, 11, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = "#854d0e";
    ctx.fillRect(c.x + 9, c.y + 4, 4, 14);
}

function draw() {

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // clouds
    ctx.fillStyle = "rgba(255,255,255,.8)";

    ctx.fillRect(100, 80, 100, 25);
    ctx.fillRect(125, 65, 50, 40);

    ctx.fillRect(700, 100, 120, 25);
    ctx.fillRect(730, 80, 60, 45);

    // platforms
    for (const p of platforms) {

        ctx.fillStyle = "#075985";
        ctx.fillRect(p.x, p.y, p.w, p.h);

        ctx.fillStyle = "#0ea5e9";
        ctx.fillRect(p.x, p.y, p.w, 6);
    }

    // coins
    for (const c of coins)
        drawCoin(c);

    // enemies
    for (const e of enemies)
        if (e.x > -50)
            drawEnemy(e);

    drawPlayer();

    // UI
    ctx.fillStyle = "#020617";
    ctx.font = "bold 24px monospace";

    ctx.fillText("PRATHMESH PIXEL RUNNER", 25, 35);
    ctx.fillText("SCORE: " + score, 25, 65);

    ctx.font = "16px monospace";
    ctx.fillText("← → MOVE   SPACE / ↑ JUMP   R RESTART", 620, 35);

    if (gameOver) {

        ctx.fillStyle = "rgba(2,6,23,.85)";
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        ctx.fillStyle = "#ffffff";
        ctx.font = "bold 52px monospace";
        ctx.fillText("GAME OVER", 330, 250);

        ctx.font = "22px monospace";
        ctx.fillText("Press R to restart", 365, 300);
    }

    if (win) {

        ctx.fillStyle = "rgba(2,6,23,.85)";
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        ctx.fillStyle = "#38bdf8";
        ctx.font = "bold 48px monospace";
        ctx.fillText("LEVEL COMPLETE!", 280, 250);

        ctx.fillStyle = "#ffffff";
        ctx.font = "22px monospace";
        ctx.fillText("Press R to play again", 360, 300);
    }
}

function loop() {

    update();
    draw();

    requestAnimationFrame(loop);
}

loop();
</script>
